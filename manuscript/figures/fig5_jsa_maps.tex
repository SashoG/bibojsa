% below in step with JSA_WINDOW in make_figure_data.py.

           mesh/rows=\csname jsarows#1\endcsname,
           point meta=explicit, forget plot]
    table [col sep=comma, x=dws_rad_ps, y=dwi_rad_ps, meta=amp]
    {\figdatadir/jsa_#1.csv};}
